\MMConfirmationSubsection*{複数段落・別行数式を含む例}

\begin{mmconfirmationgeneralhint}[1]
まず差を作る．
\[
  \sum_{k=1}^{n}(b_{k+1}-b_k)=b_{n+1}-b_1
\]
この形に持ち込めれば，中間項はすべて消える．
\end{mmconfirmationgeneralhint}

\begin{mmconfirmationadvancedcaution}[2]
$r=1$ のときは等比数列の和の公式が使えない．
\[
  S_n=\dfrac{a(1-r^{n})}{1-r}\qquad(r\neq1)
\]
分母が $0$ になる場合を必ず分けて書く．
\end{mmconfirmationadvancedcaution}

\MMConfirmationSubsection*{答の終わりと参照}

\begin{mmconfirmationproof}
$n=1$ のとき命題は成り立つ．$n=k$ で成り立つと仮定すると，
$n=k+1$ でも成り立つ．よって，すべての自然数 $n$ について成り立つ．
\MMQED
\end{mmconfirmationproof}
